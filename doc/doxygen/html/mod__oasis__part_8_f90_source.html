159  SUBROUTINE oasis_part_setup()
160  IMPLICIT NONE
161 
162  !--------------------------------------------------------
163  integer(kind=ip_intwp_p) :: m,n,k,p,nsegs,numel,taskid
164  INTEGER(kind=ip_intwp_p) :: icpl,ierr,ilen
165  integer(kind=ip_intwp_p),pointer :: start(:),length(:)
166  integer(kind=ip_intwp_p),pointer :: kparal(:)
167  integer(kind=ip_intwp_p) :: ig_size
168  integer(kind=ip_intwp_p) :: pcnt
169  logical :: found
170  character(len=ic_lvar2), allocatable :: pname0(:),pname(:)
171  logical, parameter :: local_timers_on = .false.
172  character(len=*),parameter :: subname = '(oasis_part_setup)'
173  !--------------------------------------------------------
174 
175  call oasis_debug_enter(subname)
176 
177  if (local_timers_on) then
178  call oasis_timer_start('part_setup_barrier')
179  if (mpi_comm_local /= mpi_comm_null) &
180  call mpi_barrier(mpi_comm_local, ierr)
181  call oasis_timer_stop('part_setup_barrier')
182  endif
183  if (local_timers_on) call oasis_timer_start('part_setup')
184 
185  !-----------------------------------------------
186  !> * Generate reduced partname list
187  !-----------------------------------------------
188  IF (local_timers_on) CALL oasis_timer_start('part_setup_reducelists')
189  allocate(pname0(prism_npart))
190  do n = 1,prism_npart
191  pname0(n) = prism_part(n)%partname
192  enddo
193  call oasis_mpi_reducelists(pname0,mpi_comm_local,pcnt,pname,'part_setup',fastcheck=.true.)
194  deallocate(pname0)
195  IF (local_timers_on) CALL oasis_timer_stop('part_setup_reducelists')
196 
197  !-------------------------------------------------
198  !> * Define all partitions on all tasks
199  !-------------------------------------------------
200 
201  if (local_timers_on) then
202  call oasis_timer_start('part_setup_initgsm_barrier')
203  if (mpi_comm_local /= mpi_comm_null) &
204  call mpi_barrier(mpi_comm_local, ierr)
205  call oasis_timer_stop('part_setup_initgsm_barrier')
206  endif
207  IF (local_timers_on) CALL oasis_timer_start('part_setup_initgsm')
208  do p = 1,pcnt
209 
210  if (local_timers_on) call oasis_timer_start('part_setup_initgsm_A')
211  !--- set m, either a prism_part that already exists
212  found = .false.
213  n = 0
214  do while (n < prism_npart .and. .not.found)
215  n = n + 1
216  if (prism_part(n)%partname == pname(p)) then
217  m = n
218  found = .true.
219  endif
220  enddo
221  if (local_timers_on) call oasis_timer_stop('part_setup_initgsm_A')
222 
223  if (local_timers_on) call oasis_timer_start('part_setup_initgsm_B')
224  !--- or m is a new prism_part that must be instantiated
225  !--- and set to have no data
226  if (.not.found) then
227  prism_npart = prism_npart + 1
228  m = prism_npart
229  call oasis_part_zero(prism_part(prism_npart))
230  prism_part(prism_npart)%partname = pname(p)
231  allocate(prism_part(prism_npart)%kparal(3))
232  prism_part(prism_npart)%kparal = 0
233  endif
234  if (local_timers_on) call oasis_timer_stop('part_setup_initgsm_B')
235 
236  !-------------------------------------------------
237  !> * Convert kparal information to data for the gsmap
238  !-------------------------------------------------
239 
240  if (local_timers_on) call oasis_timer_start('part_setup_initgsm_C')
241  allocate(kparal(size(prism_part(m)%kparal)))
242  kparal = prism_part(m)%kparal
243  ig_size = prism_part(m)%ig_size
244 
245  if (kparal(clim_strategy) == clim_serial) then
246  nsegs = 1
247  allocate(start(nsegs),length(nsegs))
248  start(1) = 1
249  length(1) = kparal(clim_length)
250  numel = nsegs
251  if (length(1) == 0) numel = 0
252  elseif (kparal(clim_strategy) == clim_apple) then
253  nsegs = 1
254  allocate(start(nsegs),length(nsegs))
255  start(1) = kparal(clim_offset) + 1
256  length(1) = kparal(clim_length)
257  numel = nsegs
258  if (length(1) == 0) numel = 0
259  elseif (kparal(clim_strategy) == clim_box) then
260  nsegs = kparal(clim_sizey)
261  allocate(start(nsegs),length(nsegs))
262  do n = 1,nsegs
263  start(n) = kparal(clim_offset) + (n-1)*kparal(clim_ldx) + 1
264  length(n) = kparal(clim_sizex)
265  enddo
266  numel = nsegs
267  if (kparal(clim_sizey)*kparal(clim_sizex) == 0) numel = 0
268  elseif (kparal(clim_strategy) == clim_orange) then
269  nsegs = kparal(clim_segments)
270  allocate(start(nsegs),length(nsegs))
271  numel = 0
272  DO n = 1,nsegs
273  ilen = kparal((n-1)*2 + 4)
274  IF (ilen > 0) THEN
275  numel = numel + 1
276  start(numel) = kparal((n-1)*2 + 3) + 1
277  length(numel) = ilen
278  ENDIF
279  ENDDO
280  elseif (kparal(clim_strategy) == clim_points) then
281  nsegs = kparal(clim_segments)
282  allocate(start(nsegs),length(nsegs))
283  numel = 0
284  if (nsegs > 0) then
285  !--- initialize first segment, nsegs=1,n=1,k=3
286  nsegs = 1
287  n = 1
288  k = n+2
289  start(nsegs) = kparal(k)
290  length(nsegs) = 1
291  !--- compute rest of segments from n=2,k=4
292  do n = 2,kparal(clim_segments)
293  k = n+2
294  if (kparal(k)-kparal(k-1) == 1) then
295  length(nsegs) = length(nsegs) + 1
296  else
297  nsegs = nsegs + 1
298  start(nsegs) = kparal(k)
299  length(nsegs) = 1
300  endif
301  enddo
302  numel = nsegs
303  endif
304  else
305  write(nulprt,*) subname,estr,'part strategy unknown in def_part = ',kparal(clim_strategy)
306  write(nulprt,*) subname,estr,'strategy set in kparal array index ',clim_strategy
307  call oasis_abort(file=__file__,line=__line__)
308  endif
309 
310  IF (oasis_debug >= 30) THEN
311  WRITE(nulprt,*) subname, ' Nsegs before calling mct_gsmap_init :',nsegs
312  WRITE(nulprt,*) subname, ' Numel before calling mct_gsmap_init :',numel
313  CALL oasis_flush(nulprt)
314  ENDIF
315 
316  if (local_timers_on) call oasis_timer_stop('part_setup_initgsm_C')
317 
318  !-------------------------------------------------
319  !> * Initialize the local gsmap and partition gsmap
320  !-------------------------------------------------
321 
322  if (local_timers_on) call oasis_timer_start('part_setup_initgsm_D')
323  if (mpi_comm_local /= mpi_comm_null) then
324  if (ig_size > 0) then
325  call mct_gsmap_init(prism_part(m)%gsmap,start,length,mpi_root_local,&
326  mpi_comm_local,compid,numel=numel,gsize=ig_size)
327  else
328  call mct_gsmap_init(prism_part(m)%gsmap,start,length,mpi_root_local,&
329  mpi_comm_local,compid,numel=numel)
330  endif
331  prism_part(m)%gsize = mct_gsmap_gsize(prism_part(m)%gsmap)
332  prism_part(m)%lsize = mct_gsmap_lsize(prism_part(m)%gsmap,mpi_comm_local)
333  icpl = mpi_undefined
334  if (numel > 0) icpl = 1
335  CALL mpi_comm_split(mpi_comm_local,icpl,1,prism_part(m)%mpicom,ierr)
336  if (numel > 0) then
337  CALL mpi_comm_size ( prism_part(m)%mpicom, prism_part(m)%npes, ierr )
338  CALL mpi_comm_rank ( prism_part(m)%mpicom, prism_part(m)%rank, ierr )
339  if (ig_size > 0) then
340  call mct_gsmap_init(prism_part(m)%pgsmap,start,length,0, &
341  prism_part(m)%mpicom,compid,numel=numel,gsize=ig_size)
342  else
343  call mct_gsmap_init(prism_part(m)%pgsmap,start,length,0, &
344  prism_part(m)%mpicom,compid,numel=numel)
345  endif
346  else
347  ! override mpicom created by split with null
348  prism_part(m)%mpicom = mpi_comm_null
349  endif
350  !else
351  !! set by default
352  ! prism_part(m)%gsize = -1
353  ! prism_part(m)%mpicom = MPI_COMM_NULL
354  endif
355  if (local_timers_on) call oasis_timer_stop('part_setup_initgsm_D')
356 
357  if (local_timers_on) call oasis_timer_start('part_setup_initgsm_E')
358  deallocate(start,length)
359  deallocate(kparal)
360  deallocate(prism_part(m)%kparal)
361 
362  if (oasis_debug >= 2) then
363  call oasis_part_write(prism_part(m),m)
364  endif
365  if (local_timers_on) call oasis_timer_stop('part_setup_initgsm_E')
366 
367  enddo ! p = 1,pcnt
368 
369  deallocate(pname)
370 
371  if (local_timers_on) call oasis_timer_stop ('part_setup_initgsm')
372  if (local_timers_on) call oasis_timer_stop('part_setup')
373 
374  call oasis_debug_exit(subname)
375 
376  END SUBROUTINE oasis_part_setup

675  SUBROUTINE oasis_part_create(id_part,TYPE,gsize,nx,ny,gridname,gscomm,mpicom,gridID)
676 
677  IMPLICIT NONE
678 
679  integer(ip_i4_p),intent(out) :: id_part !< partition id
680  character(len=*),intent(in) :: type !< type of decomposition specified
681  integer(ip_i4_p),intent(in) :: gsize !< global size of grid
682  integer(ip_i4_p),intent(in) :: nx !< global nx size
683  integer(ip_i4_p),intent(in) :: ny !< global ny size
684  character(len=*),intent(in) :: gridname !< grid name
685  integer(ip_i4_p),intent(in) :: gscomm !< global seg map communicator
686  integer(ip_i4_p),intent(in) :: mpicom !< local mpi comm
687  integer(ip_i4_p), optional :: gridID(:)!< gridcell ID
688  !--------------------------------------------------------
689  integer(ip_i4_p) :: gsrank
690  integer(ip_i4_p) :: gssize
691  integer(ip_i4_p) :: numel

693  integer(ip_i4_p),pointer :: llist(:),glist(:)
694  integer(ip_i4_p) :: pts
695  integer(ip_i4_p) :: found,foundall
696  integer(ip_i4_p) :: n

698  character(len=*),parameter :: subname = '(oasis_part_create)'

701  call oasis_debug_enter(subname)
702

704  call mpi_comm_rank(gscomm,gsrank,ierr)

708  gssize = -1
709  endif

711  if (oasis_debug >= 15) then
712  write(nulprt,*) subname,' called with ',gsize,nx,ny,trim(gridname)
713  write(nulprt,*) subname,' local ',gsrank,gssize
714  endif
715 
716  if ((type == 'decomp_wghtfile' .and. .not.present(gridid)) .or. &
717  (type /= 'decomp_wghtfile' .and. present(gridid))) then
718  write(nulprt,*) subname,estr,'decomp_wghtfile and gridID arguments inconsistent ',trim(type)
719  call oasis_abort(file=__file__,line=__line__)
720  endif
721 
722  !-----------------------------------------------
723  !> * Check if an existing gsmap can be reused
724  !-----------------------------------------------
725

728  n = 0

730  n = n + 1

732  trim(prism_part(n)%gridname) == trim(gridname) .and. &
733  prism_part(n)%mpicom == gscomm .and. &
734  prism_part(n)%nx == nx .and. &
735  prism_part(n)%ny == ny) then

737  found = 1
738  endif

743  !-----------------------------------------------
744 
745  foundall = -1
746  call oasis_mpi_min(found,foundall,mpicom,string=subname//' found',all=.true.)
747  if (foundall == 1) then
748  if (oasis_debug >= 2) then
749  write(nulprt,*) subname,' reuse part ',prism_npart,gsize
750  endif
751  call oasis_debug_exit(subname)
752  return
753  endif
754 
755  !-----------------------------------------------
756  !> * Instantiate a decomposition based on gsize and type
757  !-----------------------------------------------
758 
759  prism_npart = prism_npart + 1
760  call oasis_part_zero(prism_part(prism_npart))
761 
762  !-----------------------------------------------
763  !> * Create a new partition and set values
764  !-----------------------------------------------
765 
766  part_name_cnt = part_name_cnt + 1
767  write(prism_part(prism_npart)%partname,'(a,i6.6)') trim(compnm)//'_part',part_name_cnt
768  prism_part(prism_npart)%gsize = gsize
769  prism_part(prism_npart)%nx = nx
770  prism_part(prism_npart)%ny = ny
771  prism_part(prism_npart)%gridname = trim(gridname)
772  prism_part(prism_npart)%mpicom = gscomm
773  prism_part(prism_npart)%npes = gssize
774  prism_part(prism_npart)%rank = gsrank
775 
776  allocate(start(1),length(1))
777  start = 1
778  length = 0
779  numel = 0
780 
781  if (trim(type) == 'decomp_1d') then
782  pts = 0
783  if (gsrank >= 0) then
784  numel = 1
785  length(1) = gsize/gssize
786  pts = gsize - length(1)*gssize
787  if (gsrank < pts) length(1) = length(1) + 1
788  start(1) = gsize/gssize*(gsrank) + min(gsrank,pts) + 1
789  endif
790 
791  if (oasis_debug >= 15) then
792  write(nulprt,*) subname,trim(type),numel,start,length,pts
793  endif
794 
795  elseif (trim(type) == 'decomp_wghtfile') then
796  allocate(llist(gsize),glist(gsize))
797  llist = -1
798  numel = 0
799  if (gsrank >= 0) then
800  if (oasis_debug >= 15) then
801  write(nulprt,*) subname,' wgts1 ',size(gridid)
802  write(nulprt,*) subname,' gridID ',minval(gridid),maxval(gridid)
803  endif
804  do n = 1,size(gridid)
805  if (gridid(n) > 0 .and. gridid(n) <= gsize) then
806  numel = numel + 1
807  llist(gridid(n)) = gsrank
808 ! elseif (gridID(n) > gsize) then
809 ! tcraig, allow > gsize and ignore it, errors trapped/ignored in map read
810 ! write(nulprt,*) subname,estr,'gridID > gsize',gridID(n),gsize
811 ! call oasis_abort(file=__FILE__,line=__LINE__)
812 ! else
813 ! tcraig, allow <= 0 and ignore it, errors trapped/ignored in map read
814 ! write(nulprt,*) subname,estr,'gridID <= 0',gridID(n),gsize
815 ! call oasis_abort(file=__FILE__,line=__LINE__)
816  endif
817  enddo
818 
819  ! this computes the max MPI rank that includes the gridcell
820  ! max is arbitrary but this forces each gridcell to be associated with just one rank
821  call oasis_mpi_max(llist,glist,gscomm,string=trim(subname)//' glist',all=.true.)
822 
823  deallocate(llist)
824  deallocate(start,length)
825  allocate(start(numel),length(numel))
826  start = -1
827  length = -1
828  numel = 0
829  do n = 1,gsize
830  if (glist(n) == gsrank) then
831  numel = numel + 1
832  if (numel > size(gridid)) then
833  write(nulprt,*) subname,estr,'numel error ',numel,size(gridid)
834  call oasis_abort(file=__file__,line=__line__)
835  endif
836  start(numel) = n
837  length(numel) = 1
838  endif
839  enddo
840  deallocate(glist)
841  endif ! gsrank >= 0
842 
843  if (oasis_debug >= 15) then
844  write(nulprt,*) subname,trim(type),numel
845  call oasis_flush(nulprt)
846  endif
847 
848  else
849  write(nulprt,*) subname,estr,'type argument unknown = ',trim(type)
850  call oasis_abort(file=__file__,line=__line__)
851  endif
852 
853  !-----------------------------------------------
854  !> * Initialize the partition gsmap and pgsmap
855  !-----------------------------------------------
856 
857  call mct_gsmap_init(prism_part(prism_npart)%gsmap,start,length,0,mpicom,compid,gsize=gsize,numel=numel)
858  if (gsrank >= 0) then
859  call mct_gsmap_init(prism_part(prism_npart)%pgsmap,start,length,0, &
860  prism_part(prism_npart)%mpicom,compid,gsize=gsize,numel=numel)
861  endif
862  deallocate(start,length)
863  if (oasis_debug >= 2) then
864  write(nulprt,*) subname,' create new part ',prism_npart,gsize
865  call oasis_part_write(prism_part(prism_npart),prism_npart)
866  endif
867 
868  id_part = prism_npart
869 
870  call oasis_debug_exit(subname)
871 
872 END SUBROUTINE oasis_part_create
